%Question 3
\begin{question}

\begin{equation*}
f_{n}(x) = 
\begin{cases}
\sin(\pi x), \text{ if } 2n \leq x < (2n+1),\\
0, \text{ otherwise. }
\end{cases}
\end{equation*}

\qpart


\qpart

To test for pointwise limits we can take \( x = 2.5 \) as a constant value.

\begin{align*}
\stext{When \( n=1 \) the interval is \( [2,3) \)}
f_1(2.5) &= \sin(2.5\pi) = 1\\[8pt]
\stext{When \( n=2 \) the interval is \( [4,5) \)}
f_2(2.5) &= 0\\[8pt]
\stext{When \( n=3 \) the interval is \( [6,7) \)}
f_3(2.5) &= 0\\[8pt]
&\vdots \\[8pt]
\stext{When \( n=k \) the interval is \( [2k,2k+1) \)}
f_k(2.5) &= 0\\[8pt]
\end{align*}

\[ \forall x \in \R, \lim_{n\to\infty} = 0 \]

Therefore \( f_n \to f \text{ pointwise as } n \to \infty \) 

\vspace{3cm}

\qpart


Using \textup{Dfinition 4.4 Chapter 15 p 78} we can test for uniform convergence.

Assume \( \epsilon = \frac{1}{2} \)
\begin{align*}
|f_n(x) - f(x)| &< \epsilon \\[8pt]
|\sin(\pi x) - 0| &< \frac{1}{2} \\[8pt]
|\sin(\pi x)| &< \frac{1}{2} \\[8pt]
\end{align*}

This is true for some values of \( x \) but not for all values of \( x \) in the domain of \( f_n(x) \).

Therefore \( f_n \) does not converge uniformly to \( f \text{ as } n \to \infty \)

\end{question}

\begin{question}

Part c) alternative solution: 

\qpart

\begin{align*}
M_{n} &= \sup_{x \in \R} |f_n(x) - f(x)| : x \in \R \\[8pt]
&= \sup_{x \in [2n,2n+1)} |\sin(\pi x) - 0| \\[8pt]
&= \sup_{x \in [2n,2n+1)} |\sin(\pi x)| \\[8pt]
&= 1 \\[8pt]
\end{align*}

Since \( M_n = 1 \) for all \( n \), \( M_n \not\to 0 \text{ as } n \to \infty \)
Therefore \( f_n \) does not converge uniformly to \( f \text{ as } n \to \infty \)


\end{question}
